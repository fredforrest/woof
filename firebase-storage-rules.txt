rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    // Avatar uploads - users can only upload their own avatars
    match /avatars/{fileName} {
      allow read: if request.auth != null;
      allow write: if request.auth != null
        && request.resource.size < 5 * 1024 * 1024 // 5MB limit for avatars
        && request.resource.contentType.matches('image/.*')
        && fileName.matches('avatar_' + request.auth.uid + '_.*');
    }
    
    // Chat room photo uploads
    match /chatRooms/{roomId}/{fileName} {
      allow read: if request.auth != null;
      allow write: if request.auth != null
        && request.resource.size < 10 * 1024 * 1024 // 10MB limit for chat photos
        && request.resource.contentType.matches('image/.*')
        && fileName.matches('^' + request.auth.uid + '_.*');
    }
  }
}
